    p.type = p.type==="password" ? "text" : "password";

  const PRIVS = {
    "Headteacher": { icon:"👑", can: ["➕ Add / Edit Pupils","✏️ Enter / Edit Marks","📋 View Broadsheet","🎓 Print Report Cards","📈 View Analysis","⚙️ Settings","💾 Save Year Archive","🗑️ Delete Pupils / Results"] },

  };

  async function doLogin(){
    const u = document.getElementById('username').value.trim();
    const p = document.getElementById('password').value;
    const msg = document.getElementById('msg');
    if(!u || !p){ msg.textContent="⚠️ Enter username and password"; msg.style.color="red"; return; }
    msg.textContent="🔍 Checking..."; msg.style.color="#1a4d8f";
    const { data, error } = await supabaseClient.from('app_users').select('*').eq('username',u).eq('password',p).eq('active',true).maybeSingle();
    if(error || !data){ msg.textContent="❌ Wrong username or password"; msg.style.color="red"; return; }
    // Save session
    localStorage.setItem("dondi_user", JSON.stringify({ username: data.username, full_name: data.full_name, role: data.role, school: data.school }));
    if(data.school && data.school!=="Combined") localStorage.setItem("selectedSchool", data.school);
    showUser(data);
  }

  function guestLogin(){
    const guest = { username:"guest", full_name:"Guest", role:"Viewer", school:"Combined" };
    localStorage.setItem("dondi_user", JSON.stringify(guest));
    showUser(guest);
  }

  function showUser(u){
    document.getElementById('loginForm').style.display="none";
    const box = document.getElementById('userInfo');
    box.style.display="block";
    const priv = PRIVS[u.role] || PRIVS["Viewer"];
    document.getElementById('avatarIcon').textContent = priv.icon;
    document.getElementById('welcomeName').textContent = "Welcome, " + (u.full_name || u.username);
    document.getElementById('welcomeRole').textContent = u.role + " • " + (u.school || "Combined");
    let html = "<b>✅ You can:</b><br>" + priv.can.map(c=>"• "+c).join("<br>");
    if(priv.cannot) html += "<br><br><b>❌ You cannot:</b><br>" + priv.cannot.map(c=>"• "+c).join("<br>");
    document.getElementById('privList').innerHTML = html;
  }

  function goDashboard(){ window.location.href="index.html"; }
  function doLogout(){ localStorage.removeItem("dondi_user"); location.reload(); }

  // If already logged in, show info
  const saved = localStorage.getItem("dondi_user");
  if(saved){

  }
  document.getElementById('password').addEventListener('keypress', e=>{ if(e.key==="Enter") doLogin(); });
