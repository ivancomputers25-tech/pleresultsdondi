create table if not exists app_users (
  id bigint generated always as identity primary key,
  username text unique not null,
  password text not null,
  full_name text,
  role text default 'Teacher',
  school text default 'Combined',
  active boolean default true,
  created_at timestamp default now()
);
alter table app_users enable row level security;
create policy "allow all for anon" on app_users for all using (true) with check (true);

create table if not exists system_settings (
  id bigint generated always as identity primary key,
  centre_number text,
  school_a_name text,
  school_b_name text,
  current_year int default 2026,
  emis text,
  headteacher_name text,
  headteacher_username text,
  updated_at timestamp default now()
);
alter table system_settings enable row level security;
create policy "allow all for anon" on system_settings for all using (true) with check (true);

How It Works Now — Full Flow
1. *First time:* `login.html` checks Supabase → No Headteacher → Shows *Create Headteacher Account* → Saves to `app_users` + `system_settings.headteacher_name`
2. *Next time:* `login.html` checks Supabase → Headteacher exists → Shows normal Login only, creation form hidden forever
3. *Settings page:* Shows that headteacher name from `system_settings` at top
4. *Change Headteacher:* Headteacher goes to Settings → Tap *Reset Headteacher* → Type `RESET` → Account deleted → Logout → Login page shows Create form again for new headteacher. Teachers, pupils, results are safe.
*Important:* Make sure your `system_settings` table has these 2 columns. If you already created it before, run this in Supabase SQL:
alter table system_settings add column if not exists headteacher_name text;
alter table system_settings add column if not exists headteacher_username text;
